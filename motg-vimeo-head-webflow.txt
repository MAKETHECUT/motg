<script>
(function () {
  var preconnect = ['https://player.vimeo.com', 'https://i.vimeocdn.com', 'https://f.vimeocdn.com'];
  preconnect.forEach(function (href) {
    var link = document.createElement('link');
    link.rel = 'preconnect';
    link.href = href;
    link.crossOrigin = 'anonymous';
    document.head.appendChild(link);
  });

  function observeEl(iframe) {
    return (
      iframe.closest('.image-wrapper, .work-thumbnail, .video-wrapper') ||
      iframe.parentElement ||
      iframe
    );
  }

  function parseId(url) {
    var m = url && url.match(/\/video\/(\d+)/);
    return m ? m[1] : null;
  }

  function ensureThumb(iframe, embedUrl) {
    var id = parseId(embedUrl);
    if (!id) return;
    var wrapper = observeEl(iframe);
    var thumb = wrapper.querySelector('img.motg-vimeo-auto-thumb');
    if (!thumb) {
      thumb = document.createElement('img');
      thumb.className = 'motg-vimeo-auto-thumb';
      thumb.alt = '';
      thumb.decoding = 'async';
      wrapper.insertBefore(thumb, wrapper.firstChild);
    }
    if (thumb.dataset.motgThumbFor !== embedUrl) {
      thumb.dataset.motgThumbFor = embedUrl;
      thumb.src = 'https://vumbnail.com/' + id + '.jpg';
    }
  }

  function gate() {
    var vh = window.innerHeight || 800;
    document.querySelectorAll('iframe[src*="player.vimeo.com"]').forEach(function (iframe) {
      if (iframe.id === 'showreel-video') return;
      if (iframe.closest('.hero .video-visual, .vimeo-wrapper, #showreel-video')) return;
      var src = iframe.getAttribute('src');
      if (!src || src.indexOf('player.vimeo.com') === -1) return;
      var rect = observeEl(iframe).getBoundingClientRect();
      iframe.setAttribute('data-vimeo-src', src);
      ensureThumb(iframe, src);
      if (!(rect.bottom > 0 && rect.top < vh * 2.4)) iframe.removeAttribute('src');
    });
  }

  document.addEventListener('DOMContentLoaded', gate);
  if (document.readyState !== 'loading') gate();
})();
</script>
